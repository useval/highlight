#include <stdio.h>
#include <stdlib.h>
#define MAX(a, b) ((a) > (b) ? (a) : (b)) \
    /* continued */

/** A growable buffer. */
typedef struct {
    size_t len;
    size_t cap;
    uint8_t *data;
} buffer_t;

static int buffer_push(buffer_t *buf, uint8_t byte) {
    if (buf->len == buf->cap) {
        size_t cap = buf->cap ? buf->cap * 2 : 16;
        uint8_t *data = realloc(buf->data, cap);
        if (data == NULL) return -1; // out of memory
        buf->data = data;
        buf->cap = cap;
    }
    buf->data[buf->len++] = byte;
    return 0;
}

int main(int argc, char **argv) {
    buffer_t buf = {0};
    const char *msg = "hello\tworld\n";
    for (int i = 0; msg[i] != '\0'; i++) {
        buffer_push(&buf, (uint8_t)msg[i]);
    }
    printf("%zu bytes, max %d, pi %.2f\n", buf.len, MAX(3, 7), 3.14159f);
    free(buf.data);
    return 0x0;
}
