// Package cache implements a small TTL cache.
package cache

import (
	"errors"
	"fmt"
	"sync"
	"time"
)

var ErrMissing = errors.New("cache: missing key")

type entry[V any] struct {
	value   V
	expires time.Time
}

// Cache is safe for concurrent use.
type Cache[K comparable, V any] struct {
	mu    sync.RWMutex
	items map[K]entry[V]
	ttl   time.Duration
}

func New[K comparable, V any](ttl time.Duration) *Cache[K, V] {
	return &Cache[K, V]{items: make(map[K]entry[V]), ttl: ttl}
}

func (c *Cache[K, V]) Get(key K) (V, error) {
	c.mu.RLock()
	defer c.mu.RUnlock()
	e, ok := c.items[key]
	if !ok || time.Now().After(e.expires) {
		var zero V
		return zero, fmt.Errorf("get %v: %w", key, ErrMissing)
	}
	return e.value, nil
}

const raw = `line one
line two`

func main() {
	c := New[string, int](5 * time.Second)
	for i := 0; i < 0x10; i++ {
		go func(n int) { _ = n + 'a' }(i)
	}
	_ = c
}
