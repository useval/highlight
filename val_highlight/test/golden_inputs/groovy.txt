/** Prints a report of the build. */
@groovy.transform.CompileStatic
class Report {
    String title = '''Multi
line, no ${interpolation}'''
    List<Integer> sizes = [1, 2, 0x10, 3.5G as int]

    String render(Map<String, ?> info) {
        def lines = info.collect { k, v -> "$k=${v ?: 'none'}\t" }
        return """Report: $title
${lines.join('\n')}"""
    }
}

tasks.register('report') {
    doLast {
        println new Report().render(version: releaseName, code: versionCode)
        assert versionCode > 0 : "bad code"
    }
}
