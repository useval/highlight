module Stats

using LinearAlgebra, Statistics
import Base: show

export Summary, summarize, @timed_log

"""
    Summary{T<:Real}

Holds summary statistics for a sample of `T`.
"""
struct Summary{T<:Real}
    n::Int
    mean::Float64
    std::T
end

mutable struct Counter
    value::Int
end

function summarize(xs::AbstractVector{T}) where {T<:Real}
    isempty(xs) && throw(ArgumentError("empty sample"))
    μ = mean(xs)
    σ = length(xs) > 1 ? std(xs) : zero(T)
    return Summary{T}(length(xs), μ, σ)
end

increment!(c::Counter, by=1) = (c.value += by; c)

show(io::IO, s::Summary) = print(io, "Summary(n=$(s.n), mean=$(round(s.mean; digits=2)))")

macro timed_log(expr)
    quote
        local t0 = time()
        local result = $(esc(expr))
        println("took $(time() - t0) s")
        result
    end
end

end # module
