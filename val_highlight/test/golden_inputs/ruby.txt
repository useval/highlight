module Billing
  class Invoice < Base::Record
    attr_reader :id, :items
    TAX_RATE = 0.2r
    @@count = 0

    def initialize(id, items: [], currency: :usd)
      @id = id
      @items = items
      @currency = currency
      @@count += 1
    end

    def total?
      items.any? && total > 0
    end

    def total
      items.sum { |item| item[:price] * (1 + TAX_RATE) }
    end

    def self.parse(text)
      data = JSON.parse(text, symbolize_names: true)
      new(data[:id], items: data.fetch(:items, []))
    rescue JSON::ParserError => e
      warn "bad invoice: #{e.message} (#{text.length} bytes)"
      nil
    end

    def to_s = "Invoice ##{id}: #{format('%.2f', total)} #{@currency.upcase}"
  end
end
