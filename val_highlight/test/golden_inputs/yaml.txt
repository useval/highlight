# Deployment configuration
apiVersion: apps/v1
kind: Deployment
metadata:
  name: web
  labels: {app: web, tier: frontend}
spec:
  replicas: 3
  selector:
    matchLabels:
      app: web
  template:
    spec:
      containers:
        - name: web
          image: "registry.example.com/web:1.4.2"
          ports:
            - containerPort: 8080
          env:
            - name: DEBUG
              value: false
            - name: RATIO
              value: 0.75
          args: &args [--port, "8080"]
          resources:
            limits: {cpu: 500m, memory: 256Mi}
